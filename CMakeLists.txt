cmake_minimum_required(VERSION 3.16)
project(logscope VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(LOGSCOPE_SANITIZE "Включить ASan и UBSan" OFF)
if(MSVC)
  add_compile_options(/W4 /WX /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
  if(LOGSCOPE_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(logscope_core STATIC src/parse.c src/stats.c src/top.c src/window.c)
target_include_directories(logscope_core PUBLIC src)
add_executable(logscope src/main.c src/reader.c src/report.c)
target_link_libraries(logscope PRIVATE logscope_core)
add_executable(logscope_unit tests/unit.c)
target_link_libraries(logscope_unit PRIVATE logscope_core)
enable_testing()
add_test(NAME core COMMAND logscope_unit)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  file(MAKE_DIRECTORY "${CMAKE_SOURCE_DIR}/build")
  add_test(NAME contracts COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests -v)
  set_tests_properties(contracts PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" ENVIRONMENT "LOGSCOPE_BIN=$<TARGET_FILE:logscope>;PYTHONIOENCODING=utf-8")
endif()
